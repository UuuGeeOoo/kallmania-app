    18→    18→function useRecalc() {
    19→    19→  return useCallback((metrics: KpiMetric[]): KpiMetric[] => {
    20→    20→    const out = metrics.map((m) => ({ ...m }));
    21→    21→    const get = (key: string): number | null => {
    22→    22→      const m = out.find((x) => x.key === key);
    23→    23→      return m && m.actual !== null && !Number.isNaN(m.actual) ? m.actual : null;
    24→    24→    };
    25→    25→    for (const m of out) {
    26→    26→      // derived metrics
    27→    27→      if (m.key === "contactRate") {
    28→    28→        const c = get("callsConnected"), t = get("totalCallsMade");
    29→    29→        const manual = m.actual !== null && !m.auto;
    30→    30→        if (!manual) { m.actual = t !== null && t !== 0 && c !== null ? Math.round((c / t) * 10000) / 100 : null; m.auto = m.actual !== null; }
    31→    31→      }
    32→    32→      if (m.key === "conversionRate") {
    33→    33→        const leads = get("hotLeads"), warm = get("warmLeads"), c = get("callsConnected");
    34→    34→        const manual = m.actual !== null && !m.auto;
    35→    35→        if (!manual) {
    36→    36→          const totalLeads = leads !== null && warm !== null ? leads + warm : null;
    37→    37→          m.actual = totalLeads !== null && c !== null && c !== 0 ? Math.round((totalLeads / c) * 10000) / 100 : null;
    38→    38→          m.auto = m.actual !== null;
    39→    39→        }
    40→    40→      }
    41→    41→      if (m.key === "leadsGenerated") {
    42→    42→        const h = get("hotLeads"), w = get("warmLeads");
    43→    43→        const manual = m.actual !== null && !m.auto;
    44→    44→        if (!manual) { m.actual = h !== null && w !== null ? h + w : null; m.auto = m.actual !== null; }
    45→    45→      }
    46→    46→      // variance
    47→    47→      if (m.actual !== null && m.target !== null) m.variance = Math.round((m.actual - m.target) * 100) / 100;
    48→    48→      else m.variance = null;
    49→    49→    }
    50→    50→    return out;
    51→    51→  }, []);
    52→    52→}
